# BME280 Indoor Sensor Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Read a BME280 on the Pico W's I2C1 bus and expose indoor temperature, humidity and pressure as layout template fields.

**Architecture:** A new dependency-free `firmware/pico/bme280.py` driver (pure compensation functions + a small class over any object with `readfrom_mem`/`writeto_mem`). `layout.py` gains a `"sensor"` field source parallel to `"weather"`. `main.py` owns the sensor instance on its own I2C controller, refreshes a cached reading every 60 s when the active template needs it, and passes it to `layout.render()`.

**Tech Stack:** MicroPython (Pico W, rp2 port), CPython 3 stdlib `unittest` for host tests.

**Spec:** `docs/superpowers/specs/2026-09-23-bme280-sensor-design.md`

## Global Constraints

- Sensor bus: `I2C(1, freq=100000, sda=Pin(2), scl=Pin(3))` — GP2/GP3, physical pins 4/5.
- Probe addresses 0x76 then 0x77; accept only chip ID `0x60` (reject BMP280 `0x58`).
- Sensor reads do **not** take `mutation_lock` (that lock guards the module bus, I2C0, only).
- `SENSOR_UPDATE_INTERVAL = 60` seconds.
- Plausibility: temp −40..85 °C, humidity 0..100 %, pressure 300..1100 hPa; outside → failed read.
- New fields: `indoor_temp_f`, `indoor_temp_c`, `indoor_humidity` (min width 2, max natural 3), `pressure_hpa` (min width 3, max natural 4).
- Missing sensor / no reading → dashes, same as never-fetched weather.
- No pytest: host tests use stdlib `unittest`, run with `python3 -m unittest discover -s firmware/pico -p "test_*.py"` from the repo root.
- Code must run on MicroPython: no f-string-only-CPython features, no `re.finditer`, no `typing`; `struct` is available.
- Commits end with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`. Never push to `master`.

import struct
import unittest

import bme280

# Calibration from the Bosch BMP280 datasheet worked example (section 8.2),
# plus humidity coefficients taken from a real BME280.
CALIB = {
    "T1": 27504, "T2": 26435, "T3": -1000,
    "P1": 36477, "P2": -10685, "P3": 3024, "P4": 2855, "P5": 140,
    "P6": -7, "P7": 15500, "P8": -14600, "P9": 6000,
    "H1": 75, "H2": 362, "H3": 0, "H4": 313, "H5": 50, "H6": 30,
}
ADC_T = 519888
ADC_P = 415148


def _humidity_float(c, t_fine, adc_h):
    # Datasheet floating-point reference formula (BME280 section 8.1).
    h = t_fine - 76800.0
    h = (adc_h - (c["H4"] * 64.0 + c["H5"] / 16384.0 * h)) * (
        c["H2"] / 65536.0 * (1.0 + c["H6"] / 67108864.0 * h * (1.0 + c["H3"] / 67108864.0 * h))
    )
    h = h * (1.0 - c["H1"] * h / 524288.0)
    return min(max(h, 0.0), 100.0)


def _calib_blocks(c):
    block1 = struct.pack(
        "<HhhHhhhhhhhhBB",
        c["T1"], c["T2"], c["T3"],
        c["P1"], c["P2"], c["P3"], c["P4"], c["P5"], c["P6"], c["P7"], c["P8"], c["P9"],
        0,  # 0xA0, unused
        c["H1"],
    )
    h4, h5 = c["H4"] & 0xFFF, c["H5"] & 0xFFF
    block2 = struct.pack("<hB", c["H2"], c["H3"]) + bytes([
        h4 >> 4,
        (h4 & 0x0F) | ((h5 & 0x0F) << 4),
        h5 >> 4,
    ]) + struct.pack("<b", c["H6"])
    return block1, block2


def _raw_data(adc_p, adc_t, adc_h):
    return bytes([
        adc_p >> 12, (adc_p >> 4) & 0xFF, (adc_p & 0x0F) << 4,
        adc_t >> 12, (adc_t >> 4) & 0xFF, (adc_t & 0x0F) << 4,
        adc_h >> 8, adc_h & 0xFF,
    ])


class FakeI2C:
    """Register-level BME280 stand-in. `chips` maps address -> chip id."""

    def __init__(self, chips, adc_h=27000):
        self.chips = chips
        self.writes = []
        block1, block2 = _calib_blocks(CALIB)
        self.regs = {0x88: block1, 0xE1: block2, 0xF3: b"\x00", 0xF7: _raw_data(ADC_P, ADC_T, adc_h)}

    def readfrom_mem(self, addr, reg, n):
        if addr not in self.chips:
            raise OSError(5)  # EIO, what MicroPython raises on a NACK
        if reg == 0xD0:
            return bytes([self.chips[addr]])
        return self.regs[reg][:n]

    def writeto_mem(self, addr, reg, data):
        if addr not in self.chips:
            raise OSError(5)
        self.writes.append((addr, reg, bytes(data)))


class CompensateTest(unittest.TestCase):
    def test_temperature_matches_datasheet(self):
        r = bme280.compensate(CALIB, ADC_T, ADC_P, 27000)
        self.assertAlmostEqual(r["temp_c"], 25.08, places=2)

    def test_pressure_matches_datasheet(self):
        # Datasheet: 100653.27 Pa
        r = bme280.compensate(CALIB, ADC_T, ADC_P, 27000)
        self.assertAlmostEqual(r["pressure_hpa"], 1006.53, places=2)

    def test_humidity_matches_float_reference(self):
        t_fine = 128422  # datasheet value for ADC_T
        for adc_h in (20000, 27000, 32000):
            r = bme280.compensate(CALIB, ADC_T, ADC_P, adc_h)
            self.assertAlmostEqual(r["humidity"], _humidity_float(CALIB, t_fine, adc_h), delta=0.01)

    def test_humidity_clamped_to_zero(self):
        r = bme280.compensate(CALIB, ADC_T, ADC_P, 0)
        self.assertEqual(r["humidity"], 0.0)

    def test_pressure_none_when_p1_zero(self):
        calib = dict(CALIB, P1=0)
        r = bme280.compensate(calib, ADC_T, ADC_P, 27000)
        self.assertIsNone(r["pressure_hpa"])


class ParseCalibrationTest(unittest.TestCase):

        self.assertEqual(bme280.parse_calibration(*_calib_blocks(calib)), calib)

    def test_probes_0x76_first(self):
        s = bme280.BME280(FakeI2C({0x76: 0x60, 0x77: 0x60}))
        self.assertEqual(s.addr, 0x76)

    def test_falls_back_to_0x77(self):
        s = bme280.BME280(FakeI2C({0x77: 0x60}))
        self.assertEqual(s.addr, 0x77)

    def test_rejects_bmp280(self):
        with self.assertRaises(OSError):
            bme280.BME280(FakeI2C({0x76: 0x58}))

    def test_raises_when_absent(self):
        with self.assertRaises(OSError):
            bme280.BME280(FakeI2C({}))

    def test_explicit_addr(self):
        s = bme280.BME280(FakeI2C({0x76: 0x60, 0x77: 0x60}), addr=0x77)
        self.assertEqual(s.addr, 0x77)

    def test_read_triggers_forced_mode_and_compensates(self):
        i2c = FakeI2C({0x76: 0x60})
        s = bme280.BME280(i2c)
        i2c.writes.clear()
        r = s.read()
        # ctrl_hum (0xF2) must be written before ctrl_meas (0xF4) to take effect.
        self.assertEqual(i2c.writes, [(0x76, 0xF2, b"\x01"), (0x76, 0xF4, b"\x25")])
        self.assertAlmostEqual(r["temp_c"], 25.08, places=2)
        self.assertAlmostEqual(r["pressure_hpa"], 1006.53, places=2)
        self.assertAlmostEqual(r["humidity"], _humidity_float(CALIB, 128422, 27000), delta=0.01)

    def test_read_propagates_oserror(self):
        i2c = FakeI2C({0x76: 0x60})
        s = bme280.BME280(i2c)
        i2c.chips = {}  # sensor unplugged
        with self.assertRaises(OSError):
            s.read()

    unittest.main()

```python
# firmware/pico/bme280.py
#
# Minimal BME280 driver (temperature, humidity, pressure) for the indoor
# sensor on I2C1. Forced mode, x1 oversampling, Bosch datasheet integer
# compensation. Works with any object providing readfrom_mem/writeto_mem, so
# it runs under CPython in the host tests with a fake bus.

import struct
import time

try:
    _sleep_ms = time.sleep_ms
except AttributeError:  # CPython (host tests)
    def _sleep_ms(ms):
        time.sleep(ms / 1000)

CHIP_ID = 0x60
ADDRESSES = (0x76, 0x77)

_REG_CHIP_ID = 0xD0
_REG_CALIB1 = 0x88   # 0x88..0xA1, 26 bytes
_REG_CALIB2 = 0xE1   # 0xE1..0xE7, 7 bytes
_REG_CTRL_HUM = 0xF2
_REG_STATUS = 0xF3
_REG_CTRL_MEAS = 0xF4
_REG_DATA = 0xF7     # 0xF7..0xFE, 8 bytes

_CTRL_HUM_X1 = 0x01
_CTRL_MEAS_FORCED_X1 = 0x25  # osrs_t=x1, osrs_p=x1, mode=forced
_STATUS_MEASURING = 0x08
_MEASURE_TIMEOUT_MS = 50     # x1/x1/x1 takes ~10 ms max


def _s12(v):
    return v - 0x1000 if v & 0x800 else v


def parse_calibration(block1, block2):
    (t1, t2, t3, p1, p2, p3, p4, p5, p6, p7, p8, p9) = struct.unpack("<HhhHhhhhhhhh", block1[:24])
    h2, h3 = struct.unpack("<hB", block2[:3])
    e4, e5, e6 = block2[3], block2[4], block2[5]
    return {
        "T1": t1, "T2": t2, "T3": t3,
        "P1": p1, "P2": p2, "P3": p3, "P4": p4, "P5": p5,
        "P6": p6, "P7": p7, "P8": p8, "P9": p9,
        "H1": block1[25],
        "H2": h2,
        "H3": h3,
        "H4": _s12((e4 << 4) | (e5 & 0x0F)),
        "H5": _s12((e6 << 4) | (e5 >> 4)),
        "H6": struct.unpack("<b", block2[6:7])[0],
    }


def compensate(c, adc_t, adc_p, adc_h):
    # Temperature (datasheet 4.2.3, 32-bit). t_fine feeds P and H.
    var1 = (((adc_t >> 3) - (c["T1"] << 1)) * c["T2"]) >> 11
    d = (adc_t >> 4) - c["T1"]
    var2 = (((d * d) >> 12) * c["T3"]) >> 14
    t_fine = var1 + var2
    temp_c = ((t_fine * 5 + 128) >> 8) / 100

    # Pressure (64-bit version; MicroPython ints are arbitrary precision).
    var1 = t_fine - 128000
    var2 = var1 * var1 * c["P6"]
    var2 += (var1 * c["P5"]) << 17
    var2 += c["P4"] << 35
    var1 = ((var1 * var1 * c["P3"]) >> 8) + ((var1 * c["P2"]) << 12)
    var1 = (((1 << 47) + var1) * c["P1"]) >> 33
    if var1 == 0:
        pressure_hpa = None  # avoid division by zero (blank calibration)
    else:
        p = 1048576 - adc_p
        p = (((p << 31) - var2) * 3125) // var1
        var1 = (c["P9"] * (p >> 13) * (p >> 13)) >> 25
        var2 = (c["P8"] * p) >> 19
        p = ((p + var1 + var2) >> 8) + (c["P7"] << 4)
        pressure_hpa = p / 25600  # Q24.8 Pa -> hPa

    # Humidity (32-bit).
    v = t_fine - 76800
    v = ((((adc_h << 14) - (c["H4"] << 20) - (c["H5"] * v)) + 16384) >> 15) * (
        ((((((v * c["H6"]) >> 10) * (((v * c["H3"]) >> 11) + 32768)) >> 10) + 2097152) * c["H2"] + 8192) >> 14
    )
    v -= ((((v >> 15) * (v >> 15)) >> 7) * c["H1"]) >> 4
    v = min(max(v, 0), 419430400)
    humidity = (v >> 12) / 1024

    return {"temp_c": temp_c, "pressure_hpa": pressure_hpa, "humidity": humidity}


class BME280:
    def __init__(self, i2c, addr=None):
        self._i2c = i2c
        candidates = ADDRESSES if addr is None else (addr,)
        self.addr = None
        for a in candidates:
            try:
                chip_id = i2c.readfrom_mem(a, _REG_CHIP_ID, 1)[0]
            except OSError:
                continue
            if chip_id == CHIP_ID:
                self.addr = a
                break
        if self.addr is None:
            raise OSError("BME280 not found")
        self._calib = parse_calibration(
            i2c.readfrom_mem(self.addr, _REG_CALIB1, 26),
            i2c.readfrom_mem(self.addr, _REG_CALIB2, 7),
        )

    def read(self):
        i2c, addr = self._i2c, self.addr
        # ctrl_hum only takes effect after a subsequent ctrl_meas write.
        i2c.writeto_mem(addr, _REG_CTRL_HUM, bytes([_CTRL_HUM_X1]))
        i2c.writeto_mem(addr, _REG_CTRL_MEAS, bytes([_CTRL_MEAS_FORCED_X1]))
        waited = 0
        while i2c.readfrom_mem(addr, _REG_STATUS, 1)[0] & _STATUS_MEASURING:
            if waited >= _MEASURE_TIMEOUT_MS:
                raise OSError("BME280 measurement timeout")
            _sleep_ms(2)
            waited += 2
        d = i2c.readfrom_mem(addr, _REG_DATA, 8)
        adc_p = (d[0] << 12) | (d[1] << 4) | (d[2] >> 4)
        adc_t = (d[3] << 12) | (d[4] << 4) | (d[5] >> 4)
        adc_h = (d[6] << 8) | d[7]
        return compensate(self._calib, adc_t, adc_p, adc_h)
```

Note: the status bit is polled rather than sleeping a fixed 10 ms first; the measurement bit may not be set yet on the very first poll right after the write on some chips, but the data registers are shadowed and only update once a conversion completes, so the worst case is returning the previous forced measurement — acceptable at a 60 s refresh.

- [ ] **Step 2: Sensor state and `read_sensor()`**

Next to `last_weather = None` (~line 958) add:

```python
last_sensor_update = 0
last_sensor = None
sensor = None              # bme280.BME280 once detected at boot, else None
sensor_error_logged = False

# Out-of-range values mean a glitched read (loose wire, bus noise), not a
# real reading -- the BME280's rated range is -40..85 C and 300..1100 hPa.
def _sensor_plausible(r):
    return (
        -40 <= r["temp_c"] <= 85
        and 0 <= r["humidity"] <= 100
        and r["pressure_hpa"] is not None
        and 300 <= r["pressure_hpa"] <= 1100
    )

def read_sensor():
    # Returns a fresh reading dict, or None if there's no sensor or this
    # read failed. Logs once per failure streak rather than every minute.
    global sensor_error_logged
    if sensor is None:
        return None
    try:
        r = sensor.read()
        if not _sensor_plausible(r):
            raise OSError("implausible reading {}".format(r))
    except OSError as e:
        if not sensor_error_logged:
            print("BME280 read error:", e)
            sensor_error_logged = True
        return None
    sensor_error_logged = False
    return r
```
