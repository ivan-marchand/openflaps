# Timezone Selection Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Replace the hardcoded `-5 * 3600` UTC offset in `main.py` with a user-selectable named timezone (chosen on the Setup page) that handles DST automatically year after year.

**Architecture:** A new `tz.py` file on the Pico holds a compact table of ~40 zones with DST transition rules (nth-weekday-of-month style). `main.py` loads the saved zone name at startup and calls `tz.get_offset()` in the display loop each minute. Two new API endpoints let the frontend read and write the setting. The Setup page gets a new timezone section (above the API key form) with a `<select>` sorted by UTC offset.

**Tech Stack:** MicroPython (`time.mktime`, `time.localtime`), Preact + Vite, plain `fetch` API calls.

#   weekday: 0=Mon … 6=Sun
#   nth_occurrence: 1=first, 2=second, -1=last

def _rule_to_timestamp(year, month, nth, weekday, hour):
    """Return Unix timestamp (UTC) for the nth occurrence of weekday in month/year at hour.
    nth=-1 means last occurrence. Treats hour as wall-clock local standard time (accuracy
    within 1 hour of the actual transition, acceptable for a clock display)."""
    # First day of month
    t_first = time.mktime((year, month, 1, 0, 0, 0, 0, 0))
    first_wd = time.localtime(t_first)[6]  # 0=Mon, 6=Sun

    if nth > 0:
        days_ahead = (weekday - first_wd) % 7
        day = 1 + days_ahead + (nth - 1) * 7
    else:
        # Negative nth: count from end of month
        next_month = month + 1 if month < 12 else 1
        next_year  = year if month < 12 else year + 1
        t_last = time.mktime((next_year, next_month, 1, 0, 0, 0, 0, 0)) - 86400
        last_day = time.localtime(t_last)[2]
        last_wd  = time.localtime(t_last)[6]
        days_before = (last_wd - weekday) % 7
        day = last_day - days_before + (nth + 1) * 7

    return time.mktime((year, month, day, hour, 0, 0, 0, 0))


def get_offset(zone_name):
    """Return the current UTC offset in seconds for the given zone name."""
    entry = ZONES.get(zone_name)
    if entry is None:
        entry = ZONES["UTC"]

    std_offset, dst_offset, dst_start, dst_end = entry[0], entry[1], entry[2], entry[3]

    if dst_offset is None:
        return std_offset

    now  = time.time()
    year = time.localtime(now)[0]

    start_ts = _rule_to_timestamp(year, dst_start[0], dst_start[1], dst_start[2], dst_start[3])
    end_ts   = _rule_to_timestamp(year, dst_end[0],   dst_end[1],   dst_end[2],   dst_end[3])

    if start_ts < end_ts:
        # Northern hemisphere: DST between start and end
        in_dst = start_ts <= now < end_ts
    else:
        # Southern hemisphere: DST wraps around year end
        in_dst = now >= start_ts or now < end_ts

    return dst_offset if in_dst else std_offset


def zone_list():
    """Return list of {name, label, offset} dicts sorted by std offset ascending."""
    return sorted(
        [{"name": k, "label": v[4], "offset": v[0]} for k, v in ZONES.items()],
        key=lambda z: z["offset"]
    )
```

import { useState, useEffect } from 'preact/hooks'
import { api } from '../api'
import './ControlPage.css'

function formatOffset(seconds) {
  const sign = seconds >= 0 ? '+' : '-'
  const abs = Math.abs(seconds)
  const h = String(Math.floor(abs / 3600)).padStart(2, '0')
  const m = String(Math.floor((abs % 3600) / 60)).padStart(2, '0')
  return `UTC${sign}${h}:${m}`
}

export function SetupPage() {
  const [ssid, setSsid] = useState('')
  const [password, setPassword] = useState('')
  const [apiKey, setApiKey] = useState('')
  const [zones, setZones] = useState([])
  const [selectedZone, setSelectedZone] = useState('')
  const [status, setStatus] = useState('')
  const [busy, setBusy] = useState(false)

  useEffect(() => {
    api.getTimezone().then(data => {
      setZones(data.zones || [])
      setSelectedZone(data.current || '')
    }).catch(() => {})
  }, [])

  async function saveWifi(e) {
    e.preventDefault()
    if (!ssid) return
    setBusy(true)
    setStatus('Saving WiFi and rebooting...')
    try { await api.saveWifi(ssid, password) }
    catch { setStatus('Saved. Device rebooting...') }
    finally { setBusy(false) }
  }

  async function saveKey(e) {
    e.preventDefault()
    if (!apiKey) return
    setBusy(true)
    try {
      const r = await api.saveApiKey(apiKey)
      setStatus(r.ok ? 'API key saved.' : r.error || 'Error.')
      if (r.ok) setApiKey('')
    } finally { setBusy(false) }
  }

  async function saveTimezone(e) {
    e.preventDefault()
    if (!selectedZone) return
    setBusy(true)
    try {
      const r = await api.saveTimezone(selectedZone)
      setStatus(r.ok ? 'Timezone saved.' : r.error || 'Error.')
    } finally { setBusy(false) }
  }

  return (
    <div class="control-page">
      <div class="page-header">
        <a href="#/" class="setup-link" style="margin-top:0;">← BACK</a>
        <span class="page-title">SETUP</span>
      </div>

      <div class="display-card" style="margin-bottom:12px;">
        <div class="field-label">Timezone</div>
        <form onSubmit={saveTimezone}>
          <select
            value={selectedZone}
            onInput={e => setSelectedZone(e.target.value)}
            style="width:100%;margin-bottom:10px;"
          >
            {zones.map(z => (
              <option key={z.name} value={z.name}>
                ({formatOffset(z.offset)}) {z.label}
              </option>
            ))}
          </select>
          <button class="btn" type="submit" disabled={busy || !selectedZone}>
            SAVE TIMEZONE
          </button>
        </form>
      </div>

      <div class="display-card" style="margin-bottom:12px;">
        <div class="field-label">Claude API Key</div>
        <form onSubmit={saveKey}>
          <input type="password" placeholder="sk-ant-..." value={apiKey} onInput={e => setApiKey(e.target.value)} autocomplete="off" />
          <div style="margin-top:10px;">
            <button class="btn" type="submit" disabled={busy || !apiKey}>SAVE API KEY</button>
          </div>
        </form>
      </div>

      <div class="display-card">
        <div class="field-label">WiFi Network</div>
        <form onSubmit={saveWifi}>
          <input type="text" placeholder="SSID" value={ssid} onInput={e => setSsid(e.target.value)} style="margin-bottom:8px;" />
          <input type="password" placeholder="Password" value={password} onInput={e => setPassword(e.target.value)} />
          <div style="margin-top:10px;">
            <button class="btn" type="submit" disabled={busy || !ssid}>UPDATE WIFI &amp; REBOOT</button>
          </div>
        </form>
      </div>

      {status && <div class="status-line" style="margin-top:12px;">{status}</div>}
    </div>
  )
}
```
